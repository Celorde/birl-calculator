OH O HOME AI PO (TRAPEZIO DESCENDENTE SOMANESSAPORRA(TRAPEZIO DESCENDENTE PRIMEIRONUMEROPORRA, TRAPEZIO DESCENDENTE SEGUNDONUMEROPORRA))
    TRAPEZIO DESCENDENTE RESULTADODESSAPORRA;
    RESULTADODESSAPORRA = PRIMEIRONUMEROPORRA + SEGUNDONUMEROPORRA;
BORA CUMPADE RESULTADODESSAPORRA;
BIRL

OH O HOME AI PO (TRAPEZIO DESCENDENTE SUBTRAINESSAPORRA(TRAPEZIO DESCENDENTE PRIMEIRONUMEROPORRA, TRAPEZIO DESCENDENTE SEGUNDONUMEROPORRA))
    TRAPEZIO DESCENDENTE RESULTADODESSAPORRA;
    RESULTADODESSAPORRA = PRIMEIRONUMEROPORRA - SEGUNDONUMEROPORRA;
BORA CUMPADE RESULTADODESSAPORRA;
BIRL

OH O HOME AI PO (TRAPEZIO MULTIPLICANESSAPORRA(TRAPEZIO PRIMEIRONUMEROPORRA, TRAPEZIO SEGUNDONUMEROPORRA))
    TRAPEZIO RESULTADODESSAPORRA;
    RESULTADODESSAPORRA = PRIMEIRONUMEROPORRA * SEGUNDONUMEROPORRA;
    
    CE QUER VER ESSA PORRA? ("Chegou: %f", PRIMEIRONUMEROPORRA);
    
    CE QUER VER ESSA PORRA? ("Chegou: %f", SEGUNDONUMEROPORRA);
BORA CUMPADE RESULTADODESSAPORRA;
BIRL

OH O HOME AI PO (TRAPEZIO DESCENDENTE DIVIDENESSAPORRA(TRAPEZIO DESCENDENTE PRIMEIRONUMEROPORRA, TRAPEZIO DESCENDENTE SEGUNDONUMEROPORRA))
    CE QUER VER ESSA PORRA? ("Chegou: %d", PRIMEIRONUMEROPORRA);
    
    CE QUER VER ESSA PORRA? ("Chegou: %d", SEGUNDONUMEROPORRA);

    TRAPEZIO RESULTADODESSAPORRA;
    RESULTADODESSAPORRA = PRIMEIRONUMEROPORRA / SEGUNDONUMEROPORRA;
BORA CUMPADE RESULTADODESSAPORRA;
BIRL

HORA DO SHOW
    TRAPEZIO PRIMEIRONUMEROPORRA, SEGUNDONUMEROPORRA, RESULTADODESSAPORRA;
    FRANGO OPERADORNESSAPORRA;
    
    QUE QUE CE QUER MONSTRAO? ("%f", &PRIMEIRONUMEROPORRA);
    QUE QUE CE QUER MONSTRAO? ("%s", &OPERADORNESSAPORRA);
    QUE QUE CE QUER MONSTRAO? ("%f", &SEGUNDONUMEROPORRA);
    
    ELE QUE A GENTE QUER? (OPERADORNESSAPORRA == '+')
        RESULTADODESSAPORRA = AJUDA O MALUCO TA DOENTE SOMANESSAPORRA(PRIMEIRONUMEROPORRA, SEGUNDONUMEROPORRA);
    QUE NAO VAI DAR O QUE? (OPERADORNESSAPORRA == '-')
        RESULTADODESSAPORRA = AJUDA O MALUCO TA DOENTE SUBTRAINESSAPORRA(PRIMEIRONUMEROPORRA, SEGUNDONUMEROPORRA);
    QUE NAO VAI DAR O QUE? (OPERADORNESSAPORRA == '*')
        RESULTADODESSAPORRA = AJUDA O MALUCO TA DOENTE MULTIPLICANESSAPORRA(PRIMEIRONUMEROPORRA, SEGUNDONUMEROPORRA);
    QUE NAO VAI DAR O QUE? (OPERADORNESSAPORRA == '/')
        RESULTADODESSAPORRA = AJUDA O MALUCO TA DOENTE DIVIDENESSAPORRA(PRIMEIRONUMEROPORRA, SEGUNDONUMEROPORRA);
    BIRL
    
    CE QUER VER ESSA PORRA? ("O RESULTADO NESSA PORRA É: %f", RESULTADODESSAPORRA);
    BORA CUMPADE 0;
